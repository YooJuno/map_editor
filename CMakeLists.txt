cmake_minimum_required(VERSION 3.21)
project(Mapwright VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets)
include(GNUInstallDirs)
option(MAPWRIGHT_FETCH_YAML "Download yaml-cpp when not installed" ON)
find_package(yaml-cpp 0.8 QUIET)
if(NOT yaml-cpp_FOUND)
  if(NOT MAPWRIGHT_FETCH_YAML)
    message(FATAL_ERROR "Install yaml-cpp >= 0.8 or enable MAPWRIGHT_FETCH_YAML")
  endif()
  include(FetchContent)
  set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(YAML_CPP_INSTALL OFF CACHE BOOL "" FORCE)
  set(YAML_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  # yaml-cpp 0.8 declares an older CMake minimum; CMake 4 requires this override.
  if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0 AND NOT DEFINED CMAKE_POLICY_VERSION_MINIMUM)
    set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
  endif()
  FetchContent_Declare(yaml-cpp
    GIT_REPOSITORY https://github.com/jbeder/yaml-cpp.git
    GIT_TAG f7320141120f720aecc4c32be25586e7da9eb978
    GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(yaml-cpp)
endif()
add_library(mapwright_core src/mapio.cpp src/document.cpp src/canvas.cpp src/mask.cpp)
target_include_directories(mapwright_core PUBLIC src)
target_link_libraries(mapwright_core PUBLIC Qt6::Widgets yaml-cpp::yaml-cpp)
target_compile_definitions(mapwright_core PUBLIC MAPWRIGHT_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(mapwright_core PUBLIC /utf-8 /W4)
else()
  target_compile_options(mapwright_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
qt_add_executable(mapwright WIN32 MACOSX_BUNDLE src/main.cpp src/mainwindow.cpp resources/resources.qrc)
target_link_libraries(mapwright PRIVATE mapwright_core)
set_target_properties(mapwright PROPERTIES
  MACOSX_BUNDLE_BUNDLE_NAME Mapwright
  MACOSX_BUNDLE_GUI_IDENTIFIER io.mapwright.editor
  MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
  MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
  WIN32_EXECUTABLE TRUE)
if(APPLE)
  install(TARGETS mapwright BUNDLE DESTINATION .)
else()
  install(TARGETS mapwright RUNTIME DESTINATION bin)
endif()
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/mapwright)
install(DIRECTORY examples DESTINATION share/mapwright)
install(DIRECTORY docs DESTINATION share/mapwright)
install(DIRECTORY licenses DESTINATION share/mapwright)
if(UNIX AND NOT APPLE)
  install(FILES resources/mapwright.desktop DESTINATION share/applications)
  install(FILES resources/mapwright.png DESTINATION share/icons/hicolor/128x128/apps)
endif()
option(MAPWRIGHT_DEPLOY_RUNTIME "Bundle Qt runtime libraries (Qt >= 6.7)" OFF)
if(MAPWRIGHT_DEPLOY_RUNTIME)
  if(Qt6_VERSION VERSION_LESS 6.7)
    message(FATAL_ERROR "Runtime deployment requires Qt >= 6.7. Build without this option on Qt 6.4.")
  endif()
  if(UNIX AND NOT APPLE)
    set_target_properties(mapwright PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  endif()
  qt_generate_deploy_app_script(TARGET mapwright OUTPUT_SCRIPT deploy_script NO_TRANSLATIONS)
  install(SCRIPT ${deploy_script})
endif()
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(mapwright_tests tests/test_mapwright.cpp src/mainwindow.cpp resources/resources.qrc)
  target_link_libraries(mapwright_tests PRIVATE mapwright_core Qt6::Test)
  add_test(NAME mapwright_tests COMMAND mapwright_tests)
  set_tests_properties(mapwright_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  qt_add_executable(mapwright_mask_tests tests/test_masks.cpp src/mainwindow.cpp resources/resources.qrc)
  target_link_libraries(mapwright_mask_tests PRIVATE mapwright_core Qt6::Test)
  add_test(NAME mapwright_mask_tests COMMAND mapwright_mask_tests)
  set_tests_properties(mapwright_mask_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
